\section{Research question and thesis}

\paragraph{Question.} Synchronous generators (SGs) are replaced by grid-following
inverters (GFLs) unit by unit. Small-signal stability is not additive: every
proper subset of a replacement portfolio can be stable while the complete
portfolio is not. Which minimal groups of replacements are mutually
incompatible, why does the network make them so, and how does the converter
control policy move that structure?

\paragraph{Thesis (current, after the cumulant negative results).}
\begin{center}
\fbox{\parbox{0.9\textwidth}{\centering
local device changes $\;\to\;$ exact network-mediated collective closure $\;\to\;$
policy-dependent minimal failing coalitions $\;\to\;$ non-composable replacement
portfolios $\;\to\;$ exact diagnosis, planning and targeted design.}}
\end{center}

\paragraph{Three manuscript contributions (TX4).}
\begin{itemize}
\item[\textbf{C1}] \textbf{Policy-dependent minimal incompatibility.} The minimal incompatibility
hypergraph $\Hy(\theta)$ and its minimum order $\kappa(\theta)$, defined exactly in transverse
coordinates; $\Hy$ changes only at imaginary-axis crossings; a target is safe in every
implementation order iff it contains no hyperedge. On IEEE-39 the converter policy reorganizes
which coalitions fail; with documented governors the four-unit witness disappears while
policy dependence persists.
\item[\textbf{C2}] \textbf{Network-closure anatomy.} Replacement actions are affine and bus-local in the
descriptor; exact network closure turns them into principal-minor interactions of all orders;
every boundary is a zero of the interaction factor. At ten frozen boundaries no
\emph{Boolean} truncation below the cardinality of the changing coalition reaches the zero.
\item[\textbf{C3}] \textbf{Actionable boundary motion.} The port derivative gives exact boundary normals and
supports iterative (Newton) retuning; a symmetry-deflated port extends the test to
zero-frequency crossings.
\end{itemize}

\paragraph{Not claimed.} Rank explains $\kappa$ (refuted); a dominant network path explains a failure
(non-executable); a simple cycle causes a failure (refuted); nonlinear composability as a
contribution (negative result); connected cumulants as the mechanism (refuted on the flagship,
Section~\ref{sec:cc}); universality of order four (refuted).

\section{Model (frozen IEEE-39 benchmark)}\label{sec:model}

\subsection{Phasor DAE}
\begin{equation}
\dot x=f(x,z;\delta,\theta),\qquad 0=g(x,z;\delta,\theta),\qquad A=f_x-f_z g_z^{-1}g_x ,
\end{equation}
with $x$ the device states, $z\in\mathbb{R}^{2N}$ the rectangular bus voltages, $g$ Kirchhoff's
current law (bus admittance matrix, device current injections, constant-power loads), index one
($g_z$ nonsingular).

\subsection{Synchronous generators (two-axis, own base)}
\begin{align}
\dot\delta&=\omega_B(\omega-1), & M\dot\omega&=P_m-P_e-D(\omega-1),\\
T'_{d0}\dot e'_q&=e_{fd}-e'_q-(x_d-x'_d)i_d, & T'_{q0}\dot e'_d&=-e'_d+(x_q-x'_q)i_q,
\end{align}
$P_e=v_di_d+v_qi_q+r_a(i_d^2+i_q^2)$; $D=0$ and constant $P_m$ (no governor) in the frozen model.
AVR (harmonized from IEEEX1, first order): $T_E\dot e_{fd}=K_A(v_{\rm ref}+v_s-|V|)-e_{fd}$.
PSS (power-input washout plus lag): $T_6\dot x_5=P_e-x_5$, $T_4\dot x_6=T_5(P_e-x_5)/T_6-x_6$, $v_s=K_Sx_6$.
Code: \nolinkurl{src/ibr_cycles/models/ieee39_devices.py}, class \texttt{SynchronousMachine}.

\subsection{Custom grid-following inverter (10 states + 1 Q/V state)}
States $(\vartheta,x_{\rm pll},p_f,q_f,x_p,x_q,i_d,i_q,x_{id},x_{iq},x_v)$, on the device rating,
current injection $\gamma\,(i_d+ji_q)e^{j\vartheta}$ on the system base ($\gamma$ = rating/100\,MVA).
With $v_d+jv_q=Ve^{-j\vartheta}$, $P=v_di_d+v_qi_q$, $Q=v_qi_d-v_di_q$:
\begin{align}
\dot\vartheta&=k_{p,\rm pll}v_q+x_{\rm pll}, & \dot x_{\rm pll}&=k_{i,\rm pll}v_q,\\
\tau\dot p_f&=P-p_f, & \tau\dot q_f&=Q-q_f,\\
i_d^{\rm ref}&=k_p(p_{\rm ref}-p_f)+x_p, & i_q^{\rm ref}&=-\big(k_q(q_{\rm cmd}-q_f)+x_q\big),\\
\dot x_p&=k_{ip}(p_{\rm ref}-p_f), & \dot x_q&=k_{iq}(q_{\rm cmd}-q_f),\\
\dot i_d&=\tfrac{\omega_B}{x_f}\big(k_i(i_d^{\rm ref}-i_d)+x_{id}-r_fi_d\big), &
\dot i_q&=\tfrac{\omega_B}{x_f}\big(k_i(i_q^{\rm ref}-i_q)+x_{iq}-r_fi_q\big),\\
\dot x_{id}&=k_{ii}(i_d^{\rm ref}-i_d), & \dot x_{iq}&=k_{ii}(i_q^{\rm ref}-i_q),\\
q_{\rm cmd}&=k_{p,v}\,g\,(v_{\rm ref}-|V|)+x_v, & \dot x_v&=k_{i,v}\,g\,(v_{\rm ref}-|V|)-w\,(x_v-q_{\rm ref}).
\end{align}
The current-loop form above is the code form $e_{d}=v_d+k_i(i_d^{\rm ref}-i_d)+x_{id}-x_fi_q$,
$\dot i_d=(\omega_B/x_f)(e_d-v_d-r_fi_d+x_fi_q)$ after cancellation. The leak is $w=0.05$\,rad/s;
$g=0$ is exact fixed-$Q$ operation plus one decoupled pole at $-w$. No current limits, DC link,
modulation delay or ride-through logic. Code: class \texttt{GridFollowingConverter}.

\section{Theory}\label{sec:theory}

\subsection{Exact transverse quotient}
Let $R_x$ be $1$ on every rotor and PLL angle and $0$ elsewhere, and $w$ the uniform frequency shift
(speed $+1$, PLL integrator $+\omega_B$).
\begin{theorem}[Transverse quotient; \status{PROVED}, classical symmetry reduction]
If device angles enter only through $Ve^{-j\cdot}$ (rotation covariance), then $AR_x=0$. If moreover
$D=0$, $P_m$ constant and the network is frequency independent, then $Aw=\omega_BR_x$; hence
$C=\spanop\{R_x,w\}$ is $A$-invariant with Jordan block $\left[\begin{smallmatrix}0&\omega_B\\0&0\end{smallmatrix}\right]$,
and for any orthonormal $Z$ spanning $C^\perp$, $\Aperp=Z^\top AZ$,
\begin{equation}\det(sI-A)=s^2\det(sI-\Aperp).\end{equation}
The reduced field satisfies $r(x+\phi R_x)=r(x)$, $r(x+cw)=r(x)+c\,\omega_BR_x$, so
$\dot y=Z^\top r(x^*+Zy)$ is the exact nonlinear transverse system. With $D\ne0$ or governors only
$C=\spanop\{R_x\}$ remains.
\end{theorem}
The rotation is a gauge; the drift is a physical consequence of absent primary frequency control.
\emph{Transverse stability}: $\aperp(S,\theta)=\max\mathrm{Re}\,\sigma(\Aperp)<0$. No eigenvalue is
discarded by magnitude; a three-state numerical classifier labels unresolved points instead of guessing.

\subsection{Minimal incompatibility hypergraph and planning}
\begin{definition}
With a stable base, $\mathcal U(\theta)=\{S:\aperp(S,\theta)\ge0\}$; $\Hy(\theta)$ is the family of
inclusion-minimal elements of $\mathcal U$; $\kappa(\theta)=\min_{e\in\Hy}|e|$ ($\infty$ if empty).
\end{definition}
\begin{proposition}[Structure, any-order safety; \status{PROVED}, elementary]
$\Hy$ is an antichain; a portfolio containing no hyperedge is stable; $\mathcal U$ need not be upward
closed. For a target $T$: (A) $T$ stable; (B) some one-at-a-time chain is stable throughout; (C) every
$R\subseteq T$ stable. Then C$\Rightarrow$B$\Rightarrow$A, no converse in general, and $T$ satisfies C iff it
contains no hyperedge.
\end{proposition}
\begin{theorem}[Boundary localization; \status{PROVED}]
Along a continuous policy path with $g_z$ nonsingular, $\Hy$ is constant on every interval on which no
$\Aperp(S)$ has an imaginary-axis eigenvalue; every change of $\Hy$ is bracketed by such a crossing.
\end{theorem}
\textbf{Supporting results (\status{PROVED}, credited).} Monotone (Metzler, order-preserving) class
$\Rightarrow$ A$=$B$=$C (Perron--Frobenius); IEEE-39 is outside it (three robust re-stabilizing pairs).
Minimum-cardinality destabilization is NP-complete (sparse-PCA clique reduction; no priority claimed).
Resilience complex $K(\rho)$: minimal non-faces $=\Hy_{NL}(\rho)$ (Stanley--Reisner, tautological).

\subsection{Descriptor realization, network closure, principal minors}
Under matched dispatch every candidate bus carries both devices (SG weight $1-\delta_i$, GFL weight
$\delta_i$); an absent device is a stable filler.
\begin{lemma}[Affinity]
$J(\delta)=J_0+\sum_i\delta_iJ_i$ exactly, each $J_i$ bus-local. The reduced $A(\delta)$ is \emph{not} affine.
\end{lemma}
Port $T_S(s)=g_z+g_x(sI-f_x)^{-1}f_z$ (network variables kept). With $E$ selecting the $2m$ voltage
channels of candidate buses, $\Delta Y_i=E_i^\top(T_{\{i\}}-T_0)E_i$ ($2\times2$), $D=\blkdiag(\Delta Y_i)$,
$K=E^\top T_0^{-1}E=K_d+K_o$ (diagonal and off-diagonal $2\times2$ blocks),
\begin{equation} M=DK,\qquad Q=(I+DK_d)^{-1}DK_o .\end{equation}
\begin{theorem}[Network-closure factorization; \status{PROVED}, Sylvester + multilinearity]
\begin{align}
\det P_S&=h_S\det T_0\det(I+M_{SS}),\\
\det(I+M_{SS})&=\textstyle\prod_{i\in S}\det(I+M_{ii})\;\det(I+Q_{SS}),\\
\det(I+Q_{SS})&=\textstyle\sum_{T\subseteq S}\mu_T,\quad \mu_\emptyset=1,\ \mu_{\{i\}}=0,\
\mu_T=\sum_{U:\,\mathrm{blocks}(U)=T}\det Q_{UU}.
\end{align}
$\mu_T$ is the Boolean M\"obius coefficient and is invariant under block-diagonal port-basis changes.
\end{theorem}
\begin{corollary}[Boundaries are network-closure zeros]
If no member alone and no device has an eigenvalue at $s^*$: $s^*\in\sigma(\Aperp(S))$ iff
$-1\in\sigma(M_{SS}(s^*))$ iff $\det(I+Q_{SS}(s^*))=0$. With $K_o=0$ no boundary exists.
\end{corollary}
The algebraic degree of $S\mapsto\det(I+Q_{SS})$ and $\kappa$ are independent quantities.

\subsection{Boundary motion (port derivative)}
\begin{proposition}[Boundary sensitivity; \status{PROVED}, implicit function theorem]
If $-1$ is a simple eigenvalue of $M_{SS}(s^*,\theta^*)$ with right/left eigenvectors $q,p$,
\begin{equation}\frac{\partial s^*}{\partial\theta}=-\frac{p^H\partial_\theta M_{SS}\,q}{p^H\partial_sM_{SS}\,q},\end{equation}
and $\nabla_\theta\mathrm{Re}\,s^*$ is the boundary normal. Equivalently, at a simple zero of
$F(H)=\det(I+Q_{HH})$: $\partial s^*/\partial a=-\partial_aF/\partial_sF$, with
$\partial_aF=\mathrm{tr}(\mathrm{adj}(I+Q)\,\partial_aQ)$ valid at the singular point.
\end{proposition}

\subsection{Symmetry-deflated zero-frequency port}
$\det T_S(0)$ vanishes structurally. Relocate $f_x^\#=f_x-\beta UU^\top$ ($U=R_x/\|R_x\|$), then deflate the
Jordan partner $x_0=w+aR_x$: $f_x^{\#\#}=f_x^\#-\beta_2x_0x_0^\top/\|x_0\|^2$, $T^{\#\#}(0)=g_z-g_x(f_x^{\#\#})^{-1}f_z$.
\begin{theorem}[\status{PROVED}; Brauer + Schur]
$\sigma(A^{\#\#})=\sigma(\Aperp)\cup\{-\beta,-\beta_2\}$; $\det T^{\#\#}(0)=\det g_z\det A^{\#\#}/\det f_x^{\#\#}$;
its sign flips iff the parity of positive real transverse eigenvalues changes. It does not see complex
crossings or real-pair coalescences.
\end{theorem}

\subsection{Finite-amplitude connected cumulants (secondary layer)}\label{sec:cc}
For $F_s(S)=\det(I+Q_{SS}(s))$, $F(\emptyset)=1$, and $\Pi(S)$ the set partitions of $S$:
\begin{equation}\chi_S=\sum_{\pi\in\Pi(S)}(-1)^{|\pi|-1}(|\pi|-1)!\prod_{B\in\pi}F(B).\end{equation}
All statements \status{PROVED} in \nolinkurl{theory/FINITE_AMPLITUDE_CONNECTED_CUMULANTS.md}:
\begin{itemize}
\item \textbf{C1 inversion:} $F(S)=\sum_{\pi}\prod_{B\in\pi}\chi_B$, unique (exponential formula in $\mathbb C[x]/(x_i^2)$).
\item \textbf{C2:} $F$ factorizes across $\{S_1,S_2\}$ iff all mixed cumulants vanish; $\chi_S=0$ alone does not imply factorization.
\item \textbf{Connectivity:} if $F(H)=0$ and $F(B)\ne0$ for all proper $B$, the support $\{B\subseteq H:|B|\ge2,\chi_B\ne0\}$ is connected and spans $H$; boundary identity $\chi_H(s^*)=-C_H(s^*)$, $C_H=F(H)-\chi_H$.
\item \textbf{C3 scalar ports:} $\chi_S=(-1)^{|S|-1}\sum_{\gamma}\prod_{(i,j)\in\gamma}q_{ij}$ over spanning directed cycles for $|S|\ge2$ ($\chi_{\{i\}}=1+q_{ii}$).
\item \textbf{C3 $2\times2$ ports:} the scalar formula does not transfer; $\chi_S$ is the sum over block-connected channel permutations; $\chi_{ab}=-\mathrm{tr}(Q_{ab}Q_{ba})+\det Q_{ab}\det Q_{ba}$. The naive holonomy sum is only the infinitesimal cumulant.
\item \textbf{C4:} Boolean degree $d_{\rm alg}$, connected order $d_{\rm conn}$ and $\kappa$ are logically independent (ring, path, star, tuned, ring+pair constructions).
\item \textbf{Connected share} at a zero: $\nu_H=|\chi_H|/\sum_\pi|\prod\chi_B|\le1/2$; deletion displacement $\delta_B=|\chi_BF(H\setminus B)|/|\partial_sF(H)|$.
\end{itemize}
\textbf{IEEE-39 verdict (preregistered):} the flagship boundary is \emph{composite}
($\nu=0.069<0.10$; deleting $\chi_{30,33,35,37}$ moves the zero by $0.024$\,s$^{-1}<0.05$); triple boundaries
are connected at 0.10 (not at 0.20). Hence $\kappa=4$ is \emph{not} an irreducible four-way interaction.
Connected cumulants are a mathematically valid, diagnostically interesting, practically limited layer.

\subsection{Second-order consistency (methodological support)}
A second-order reduced response is consistent only if the algebraic-manifold (KCL) curvature
$f_zD^2\psi$ is retained; dropping it breaks the rotation/drift symmetry and makes the frequency
prediction about $10^3$ times worse than linear (FC07 v2). Taylor consistency, not new.
