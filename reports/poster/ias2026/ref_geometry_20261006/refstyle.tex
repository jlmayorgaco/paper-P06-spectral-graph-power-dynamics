% ---- generic small pieces -----------------------------------------------------------------
\newcommand{\GoldNumber}[4]{% x, y (centre), diameter in px, number
  \fill[PGold] (#1,#2) circle (#3/2);
  \draw[PGoldDeep,line width=0.6pt] (#1,#2) circle (#3/2-0.4);
  \node[font=\PanelFont\fontsize{29pt}{29pt}\selectfont,text=PTitle] at (#1,#2+0.3) {#4};}
\newcommand{\StepNumber}[3]{% x, y, number : dark-green disc with gold ring (inner step markers)
  \fill[PGreen] ({#1},{#2}) circle (9.6);
  \draw[PGold,line width=1.8pt] ({#1},{#2}) circle (8.9);
  \node[font=\CondBold\fontsize{24pt}{24pt}\selectfont,text=white] at ({#1},{#2+0.3}) {#3};}

% \RefPanel{n}{TITLE}{x0/y0/x1/y1}{bar height/circle diameter}{title x start/max title width (px)}
\newcommand{\RefPanel}[5]{%
  \foreach \xa/\ya/\xb/\yb in {#3}{\foreach \bh/\dm in {#4}{\foreach \xs/\tw in {#5}{%
  \draw[PBorder,line width=0.7pt,fill=PBody,rounded corners=1.2pt] (\xa,\ya) rectangle (\xb,\yb);
  \fill[PGreen,rounded corners=1.5pt] (\xa,\ya) rectangle (\xb,\ya+\bh);
  \fill[PGreen] (\xa,\ya+\bh-4) rectangle (\xb,\ya+\bh);
  \draw[PGold,line width=0.9pt] (\xa+1.5,\ya) -- (\xb-1.5,\ya);
  \GoldNumber{\xa+\dm/2+5}{\ya+\bh/2}{\dm}{#1}
  \node[anchor=west,inner sep=0pt,text=white] at (\xs,\ya+\bh/2+0.3) {\fontsize{37pt}{37pt}\selectfont\FitTitle{\tw}{#2}};}}}}
